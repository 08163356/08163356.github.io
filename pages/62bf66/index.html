import {driver} from "driver.js";
import {commonConfig} from "@/utils/userGuide/config";
import {parallelEditUserGuideStep, parallelEditUserGuideType} from "@/utils/userGuide/parallelEditGuide";
import "driver.js/dist/driver.css";
import './index.css'
import {getCurrUserNavNeeded, setCurrUserNavOff} from "@/api/userGuide";
import parallelEditPersonsStore from "@/store/parallelEdit/parallelEditPersons";
import {parallelEditDesc} from "@/utils/userGuide/descriptionGetter";
export class UserGuide {
    private currStep: number = 0
    public driverObj: any = {}
    private guideName = ''
    private guideDetail = ''
    private otherParams: any
    private currGuideType: string
    private judge = { // todo
        guideFeature: (type: string) => {
            if(type.includes('PE_') && type in parallelEditUserGuideStep) {
                return {
                    feature: "parallelEdit",
                    page: type.replace('PE_', ''),
                    isNeedGuide: this.otherParams
                }
            }
        },
        isNoNeedGuide: async (otherCondition: boolean = false) => {
            // this.dontShowAgain()
            const list = await getCurrUserNavNeeded()
            const res = list.data.guide_details && list.data.guide_details[this.guideName] && list.data.guide_details[this.guideName].includes(this.guideDetail)
            return res
        },

        isParallelEdit: false
    }

    constructor(type: string, otherParams: any = 100) {
        this.otherParams = otherParams
        this.currGuideType = type
        const guideInfo = this.judge.guideFeature(type)
        if(guideInfo) {
            this.guideName = guideInfo.feature
            this.guideDetail = guideInfo.page
            this.init(type)
        }
    }
    async init(type: string){
        if(await this.judge.isNoNeedGuide()) { return }
        switch (this.guideName){
            case 'parallelEdit':
                this.handleParallelEdit()
                break
        }
    }

    handleParallelEdit = () => {
        if(this.guideDetail === "editPage" && !this.otherParams) { return }
        const options = {
            // @ts-ignore
            steps: parallelEditUserGuideStep[this.currGuideType],
            popoverClass: 'driverjs-theme',
            ...commonConfig.driverConfig,
            onDestroyed: ()=> {
                this.dontShowAgain()
            }
        }
        if(this.currGuideType === parallelEditUserGuideType.COMMIT_CONFIG) {
            options.steps[0].popover.description = parallelEditDesc.handleConflictBtn("handleMultiConflictBtn")
        }
        // @ts-ignore
        this.driverObj = driver(options)
        this.driverObj.drive()
    }

    /**
     * @description 后续不再显示
     */
    dontShowAgain(){
        setCurrUserNavOff({
            guide_name: this.guideName,
            guide_detail: this.guideDetail
        })
        this.destroy()
    }

    destroy() { driver().destroy() }
}
